FASE 3 — Paso 3: Root module (main.tf) y Resource Groups

  # PREREQUISITO (una sola vez): Asignar Storage Blob Data Contributor al usuario
  # actual sobre el Storage Account del tfstate. Sin esto, "terraform plan" cuelga
  # al adquirir el state lock (el rol Owner en la suscripción no da acceso al
  # plano de datos del Storage Account — son planos separados en Azure).
  $me = (az ad signed-in-user show --query id --output tsv)
  az role assignment create `
    --assignee-object-id $me `
    --assignee-principal-type User `
    --role "Storage Blob Data Contributor" `
    --scope "/subscriptions/1989a8c1-76c1-4555-a228-c95f86c66635/resourceGroups/rg-platform/providers/Microsoft.Storage/storageAccounts/stlztf86c66635"

  # Aplicar formato automático
  terraform fmt -recursive

  # Verificar sintaxis
  terraform validate

  # Generar el plan de ejecución y guardarlo en un archivo binario
  # -out=tfplan: guarda el plan para aplicarlo exactamente igual después
  # Resultado: 3 to add, 0 to change, 0 to destroy
  #   + rg-network-hub-lz-dev
  #   + rg-spoke-app-lz-dev
  #   + rg-shared-lz-dev
  terraform plan -out=tfplan

  # Aplicar exactamente el plan guardado (no vuelve a calcular ni pide confirmación)
  # Resultado: Apply complete! Resources: 3 added, 0 changed, 0 destroyed.
  terraform apply "tfplan"
